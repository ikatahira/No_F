
%diagrama de classes

@startuml

class Usuario {
  -id: int
  -nome: String
  -login: String
  -senha: String
  +realizarLogin()
}

class Cliente {
  -id: int
  -nome: String
  -telefone: String
  -saldo: double
  +cadastrar()
  +editar()
  +excluir()
  +consultarSaldo()
}

class Venda {
  -id: int
  -data: Date
  -valor: double
  -pago: boolean
  +registrarVendaFiado()
}

class Pagamento {
  -id: int
  -data: Date
  -valor: double
  +realizarPagamento()
}

class Cobranca {
  -id: int
  -valor: double
  -dataEnvio: Date
  +enviarCobranca()
  +receberCobranca()
}

class Notificacao {
  -id: int
  -mensagem: String
  -data: Date
  +gerarNotificacao()
  +enviar()
}

class Relatorio {
  +gerarRelatorio()
}

class SistemaExterno {
  +enviarMensagem()
}

' RELACIONAMENTOS

Usuario --> Cliente : gerencia
Cliente "1" --> "0..*" Venda
Cliente "1" --> "0..*" Pagamento
Cliente "1" --> "0..*" Cobranca

Venda --> Cliente
Pagamento --> Cliente
Pagamento --> Venda

Cobranca --> Cliente
Cobranca --> Notificacao

Notificacao --> SistemaExterno

Relatorio --> Cliente
Relatorio --> Venda
Relatorio --> Pagamento

@enduml


% diagrama de caso de Usuario
@startuml
left to right direction

actor Comerciante
actor Cliente
actor "Sistema Externo\n(WhatsApp)" as SistemaExterno

rectangle Sistema_NO_F {

  Comerciante --> (Realizar Login)
  Comerciante --> (Cadastrar Cliente)
  Comerciante --> (Editar Cliente)
  Comerciante --> (Excluir Cliente)
  Comerciante --> (Registrar Pagamento)
  Comerciante --> (Consultar Saldo do Cliente)
  Comerciante --> (Gerar Relatórios)

  (Registrar Venda Fiado) --> (Calcular Saldo)
  (Registrar Pagamento) --> (Calcular Saldo)

  (Calcular Saldo) --> (Gerar Notificação)

  (Gerar Notificação) --> (Enviar Cobrança)

  (Enviar Cobrança) --> SistemaExterno

  Cliente --> (Receber Cobrança)
  (Receber Cobrança) --> (Realizar Pagamento)

}

@enduml

%diagrama de sequência
@startuml
start

:Realizar Login_NO_F;

:Selecionar Cliente;

if (Cliente existe?) then (Sim)
  :Registrar Venda Fiado;
else (Não)
  :Cadastrar Cliente;
  :Registrar Venda Fiado;
endif

:Calcular Saldo;

if (Saldo > 0?) then (Sim)
  :Gerar Notificação;
  :Enviar Cobrança (WhatsApp);
endif

:Cliente recebe cobrança;

if (Cliente paga?) then (Sim)
  :Registrar Pagamento;
  :Calcular Saldo;
else (Não)
  :Aguardar pagamento;
endif

stop
@enduml
